    // bloque

    // bloque

FIN HACER

    // bloque
FIN SI

    // bloque

    // bloque

// MATRICES
// declaration tipe REAL 
REAL matriz[filas][columnas]

// declaration tipe ENTERO
ENTERO matriz[filas][columnas]

// declaration tipe CADENA
CADENA matriz[filas][columnas]

// declaration tipe BOOLEANO
BOOLEANO matriz[filas][columnas]

// declaration tipe CARACTER
CARACTER matriz[filas][columnas]

// declaration y asignacion
REAL matriz[filas][columnas] = [ [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...]]

// declaration y asignacion tipe ENTERO
ENTERO matriz[filas][columnas] = [ [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...]]

// declaration y asignacion tipe CADENA
CADENA matriz[filas][columnas] = [ [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...]]

// declaration y asignacion tipe BOOLEANO
BOOLEANO matriz[filas][columnas] = [ [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...]]

// declaration y asignacion tipe CARACTER
CARACTER matriz[filas][columnas] = [ [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...], [valor1, valor2, valor3, ...]]

matriz[indice_fila][indice_columna]
